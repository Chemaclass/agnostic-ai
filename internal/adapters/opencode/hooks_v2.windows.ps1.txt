$ErrorActionPreference = 'Stop'
try {
  if ($ExecutionContext.SessionState.LanguageMode -ne 'FullLanguage') { throw 'Windows hooks require PowerShell FullLanguage and public .NET reflection' }
  $assemblyName = New-Object System.Reflection.AssemblyName('AgnosticHookOwner')
  $assembly = [AppDomain]::CurrentDomain.DefineDynamicAssembly($assemblyName, [Reflection.Emit.AssemblyBuilderAccess]::Run)
  $module = $assembly.DefineDynamicModule('AgnosticHookOwner')

  function New-Structure([string]$Name, [Collections.Specialized.OrderedDictionary]$Fields) {
    $builder = $module.DefineType($Name, [Reflection.TypeAttributes]'Public, Sealed, SequentialLayout', [ValueType])
    foreach ($field in $Fields.Keys) { [void]$builder.DefineField($field, $Fields[$field], [Reflection.FieldAttributes]::Public) }
    return $builder.CreateType()
  }
  $startupType = New-Structure 'AgnosticStartupInfo' ([ordered]@{
    cb=[int]; reserved=[IntPtr]; desktop=[IntPtr]; title=[IntPtr]
    x=[int]; y=[int]; xs=[int]; ys=[int]; xc=[int]; yc=[int]; fill=[int]; flags=[int]
    show=[int16]; reservedSize=[int16]; reservedData=[IntPtr]; input=[IntPtr]; output=[IntPtr]; error=[IntPtr]
  })
  $startupExType = New-Structure 'AgnosticStartupInfoEx' ([ordered]@{ info=$startupType; attributes=[IntPtr] })
  $processType = New-Structure 'AgnosticProcessInfo' ([ordered]@{ process=[IntPtr]; thread=[IntPtr]; pid=[int]; tid=[int] })
  $basicLimitType = New-Structure 'AgnosticBasicLimit' ([ordered]@{
    processTime=[int64]; jobTime=[int64]; flags=[uint32]; minWorking=[UIntPtr]; maxWorking=[UIntPtr]
    active=[uint32]; affinity=[UIntPtr]; priority=[uint32]; scheduling=[uint32]
  })
  $ioType = New-Structure 'AgnosticIoCounters' ([ordered]@{ r=[uint64]; w=[uint64]; o=[uint64]; rb=[uint64]; wb=[uint64]; ob=[uint64] })
  $limitType = New-Structure 'AgnosticExtendedLimit' ([ordered]@{
    basic=$basicLimitType; io=$ioType; processMemory=[UIntPtr]; jobMemory=[UIntPtr]; peakProcess=[UIntPtr]; peakJob=[UIntPtr]
  })
  $pointerSize = [IntPtr]::Size
  $expected = if ($pointerSize -eq 8) { @(104,112,24,144) } else { @(68,72,16,112) }
  $structures = @($startupType,$startupExType,$processType,$limitType)
  for ($index=0; $index -lt $structures.Length; $index++) {
    if ([Runtime.InteropServices.Marshal]::SizeOf([type]$structures[$index]) -ne $expected[$index]) { throw ('unexpected native structure size: ' + $structures[$index].Name) }
  }
  if ([Runtime.InteropServices.Marshal]::OffsetOf($startupType,'input').ToInt32() -ne ($expected[0]-3*$pointerSize)) { throw 'unexpected STARTUPINFO stdio layout' }

  $nativeBuilder = $module.DefineType('AgnosticNative', [Reflection.TypeAttributes]'Public, Abstract, Sealed')
  $importType = [Runtime.InteropServices.DllImportAttribute]
  $importConstructor = $importType.GetConstructor([type[]]@([string]))
  $importFields = [Reflection.FieldInfo[]]@(
    $importType.GetField('EntryPoint'), $importType.GetField('CharSet'), $importType.GetField('ExactSpelling'),
    $importType.GetField('SetLastError'), $importType.GetField('CallingConvention')
  )
  function Add-NativeMethod([string]$Name, [type]$Result, [type[]]$Parameters) {
    $method = $nativeBuilder.DefinePInvokeMethod($Name,'kernel32.dll',$Name,[Reflection.MethodAttributes]'Public, Static, PinvokeImpl',[Reflection.CallingConventions]::Standard,$Result,$Parameters,[Runtime.InteropServices.CallingConvention]::Winapi,[Runtime.InteropServices.CharSet]::Unicode)
    $values = [object[]]@($Name,[Runtime.InteropServices.CharSet]::Unicode,$true,$true,[Runtime.InteropServices.CallingConvention]::Winapi)
    $attribute = New-Object Reflection.Emit.CustomAttributeBuilder($importConstructor,[object[]]@('kernel32.dll'),$importFields,$values)
    $method.SetCustomAttribute($attribute)
    $method.SetImplementationFlags([Reflection.MethodImplAttributes]::PreserveSig)
    $outputParameter = switch ($Name) {
      'CreateProcessW' { 10 }
      'DuplicateHandle' { 4 }
      'GetExitCodeProcess' { 2 }
      'ReadFile' { 4 }
    }
    if ($null -ne $outputParameter) { [void]$method.DefineParameter($outputParameter,[Reflection.ParameterAttributes]::Out,'result') }
    if ($Name -eq 'ReadFile') { [void]$method.DefineParameter(2,[Reflection.ParameterAttributes]'In, Out','buffer') }
  }
  Add-NativeMethod 'CreateJobObjectW' ([IntPtr]) @([IntPtr],[IntPtr])
  Add-NativeMethod 'SetInformationJobObject' ([bool]) @([IntPtr],[int],$limitType.MakeByRefType(),[int])
  Add-NativeMethod 'InitializeProcThreadAttributeList' ([bool]) @([IntPtr],[int],[int],[IntPtr].MakeByRefType())
  Add-NativeMethod 'UpdateProcThreadAttribute' ([bool]) @([IntPtr],[uint32],[IntPtr],[IntPtr],[IntPtr],[IntPtr],[IntPtr])
  Add-NativeMethod 'DeleteProcThreadAttributeList' ([void]) @([IntPtr])
  Add-NativeMethod 'CreateProcessW' ([bool]) @([string],[Text.StringBuilder],[IntPtr],[IntPtr],[bool],[uint32],[IntPtr],[string],$startupExType.MakeByRefType(),$processType.MakeByRefType())
  Add-NativeMethod 'ResumeThread' ([uint32]) @([IntPtr])
  Add-NativeMethod 'WaitForSingleObject' ([uint32]) @([IntPtr],[uint32])
  Add-NativeMethod 'GetExitCodeProcess' ([bool]) @([IntPtr],[uint32].MakeByRefType())
  Add-NativeMethod 'TerminateJobObject' ([bool]) @([IntPtr],[uint32])
  Add-NativeMethod 'TerminateProcess' ([bool]) @([IntPtr],[uint32])
  Add-NativeMethod 'DuplicateHandle' ([bool]) @([IntPtr],[IntPtr],[IntPtr],[IntPtr].MakeByRefType(),[uint32],[bool],[uint32])
  Add-NativeMethod 'GetCurrentProcess' ([IntPtr]) @()
  Add-NativeMethod 'GetStdHandle' ([IntPtr]) @([int])
  Add-NativeMethod 'CreateFileW' ([IntPtr]) @([string],[uint32],[uint32],[IntPtr],[uint32],[uint32],[IntPtr])
  Add-NativeMethod 'CloseHandle' ([bool]) @([IntPtr])
  Add-NativeMethod 'ReadFile' ([bool]) @([IntPtr],[byte[]],[uint32],[uint32].MakeByRefType(),[IntPtr])
  $nativeType = $nativeBuilder.CreateType()
  $nativeMethods = @{}
  foreach ($method in $nativeType.GetMethods([Reflection.BindingFlags]'Public, Static, DeclaredOnly')) { $nativeMethods[$method.Name] = $method }

  function Invoke-Native([string]$Name, [object[]]$Parameters) {
    $value = $nativeMethods[$Name].Invoke($null,$Parameters)
    $lastError = [Runtime.InteropServices.Marshal]::GetLastWin32Error()
    return @{ Value=$value; Error=$lastError }
  }
  function New-NativeError([int]$Code, [string]$Operation) {
    $systemError = New-Object ComponentModel.Win32Exception($Code)
    $message = $Operation + ' (Win32 error ' + $Code + '): ' + $systemError.Message
    return (New-Object ComponentModel.Win32Exception($Code,$message))
  }
  function Assert-Native($Result, [string]$Operation) {
    if (!$Result.Value) { throw (New-NativeError $Result.Error $Operation) }
  }
  function Copy-Handle([IntPtr]$Source) {
    $parameters = [object[]]@($currentProcess,$Source,$currentProcess,[IntPtr]::Zero,[uint32]0,$true,[uint32]2)
    Assert-Native (Invoke-Native 'DuplicateHandle' $parameters) 'DuplicateHandle'
    return $parameters[3]
  }
  $currentProcess = (Invoke-Native 'GetCurrentProcess' @()).Value
  $stdinHandle = (Invoke-Native 'GetStdHandle' @(-10)).Value
  function Read-Exact([int]$Count) {
    $result = New-Object byte[] $Count
    $offset = 0
    while ($offset -lt $Count) {
      $part = New-Object byte[] ($Count-$offset)
      $parameters = [object[]]@($stdinHandle,$part,[uint32]$part.Length,[uint32]0,[IntPtr]::Zero)
      Assert-Native (Invoke-Native 'ReadFile' $parameters) 'ReadFile control'
      $read = [int]$parameters[3]
      if ($read -eq 0) { throw 'incomplete launcher control' }
      [Buffer]::BlockCopy($part,0,$result,$offset,$read)
      $offset += $read
    }
    return ,$result
  }
  $utf8 = New-Object Text.UTF8Encoding($false,$true)
  function Read-Field {
    $size = [BitConverter]::ToInt32((Read-Exact 4),0)
    if ($size -lt 0 -or $size -gt 1048576) { throw 'invalid launcher field length' }
    $value = $utf8.GetString((Read-Exact $size))
    if ($value.Contains([char]0)) { throw 'launcher process field must not contain NUL' }
    return $value
  }
  function Quote-Argument([string]$Value) {
    $result = New-Object Text.StringBuilder('"')
    $slashes = 0
    foreach ($character in $Value.ToCharArray()) {
      if ($character -eq '\') { $slashes++; continue }
      if ($character -eq '"') { [void]$result.Append([char]'\',2*$slashes+1) }
      else { [void]$result.Append([char]'\',$slashes) }
      [void]$result.Append($character)
      $slashes = 0
    }
    [void]$result.Append([char]'\',2*$slashes)
    [void]$result.Append('"')
    return $result.ToString()
  }
  $application = Read-Field
  $cwd = Read-Field
  $count = [BitConverter]::ToInt32((Read-Exact 4),0)
  if ($count -lt 0 -or $count -gt 4096) { throw 'invalid launcher argument count' }
  $command = New-Object Text.StringBuilder((Quote-Argument $application))
  for ($index=0; $index -lt $count; $index++) { [void]$command.Append(' ').Append((Quote-Argument (Read-Field))) }
  $job = $attributes = $handles = $jobs = $inputHandle = $outputHandle = $errorHandle = $nul = [IntPtr]::Zero
  $process = [Activator]::CreateInstance($processType)
  $attributesReady = $false
  try {
    $created = Invoke-Native 'CreateJobObjectW' @([IntPtr]::Zero,[IntPtr]::Zero)
    if ($created.Value -eq [IntPtr]::Zero) { throw (New-NativeError $created.Error 'CreateJobObjectW') }
    $job = $created.Value
    $basicLimit = [Activator]::CreateInstance($basicLimitType)
    $basicLimit.flags = [uint32]0x2000
    $limits = [Activator]::CreateInstance($limitType)
    $limits.basic = $basicLimit
    Assert-Native (Invoke-Native 'SetInformationJobObject' @($job,9,$limits,[Runtime.InteropServices.Marshal]::SizeOf($limitType))) 'SetInformationJobObject kill-on-close'
    $inputHandle = Copy-Handle $stdinHandle
    $errorHandle = Copy-Handle ((Invoke-Native 'GetStdHandle' @(-12)).Value)
    $created = Invoke-Native 'CreateFileW' @('NUL',[uint32]0x40000000,[uint32]3,[IntPtr]::Zero,[uint32]3,[uint32]0,[IntPtr]::Zero)
    if ($created.Value -eq [IntPtr](-1)) { throw (New-NativeError $created.Error 'CreateFileW NUL') }
    $nul = $created.Value
    $outputHandle = Copy-Handle $nul
    $parameters = [object[]]@([IntPtr]::Zero,2,0,[IntPtr]::Zero)
    $sized = Invoke-Native 'InitializeProcThreadAttributeList' $parameters
    $size = [IntPtr]$parameters[3]
    if ($size -eq [IntPtr]::Zero) { throw (New-NativeError $sized.Error 'InitializeProcThreadAttributeList size') }
    $attributes = [Runtime.InteropServices.Marshal]::AllocHGlobal($size)
    $parameters[0] = $attributes
    Assert-Native (Invoke-Native 'InitializeProcThreadAttributeList' $parameters) 'InitializeProcThreadAttributeList'
    $attributesReady = $true
    $handles = [Runtime.InteropServices.Marshal]::AllocHGlobal(3*$pointerSize)
    [Runtime.InteropServices.Marshal]::WriteIntPtr($handles,0,$inputHandle)
    [Runtime.InteropServices.Marshal]::WriteIntPtr($handles,$pointerSize,$outputHandle)
    [Runtime.InteropServices.Marshal]::WriteIntPtr($handles,2*$pointerSize,$errorHandle)
    Assert-Native (Invoke-Native 'UpdateProcThreadAttribute' @($attributes,[uint32]0,[IntPtr]0x20002,$handles,[IntPtr](3*$pointerSize),[IntPtr]::Zero,[IntPtr]::Zero)) 'UpdateProcThreadAttribute handle list'
    $jobs = [Runtime.InteropServices.Marshal]::AllocHGlobal($pointerSize)
    [Runtime.InteropServices.Marshal]::WriteIntPtr($jobs,$job)
    Assert-Native (Invoke-Native 'UpdateProcThreadAttribute' @($attributes,[uint32]0,[IntPtr]0x2000D,$jobs,[IntPtr]$pointerSize,[IntPtr]::Zero,[IntPtr]::Zero)) 'UpdateProcThreadAttribute job list'
    $startup = [Activator]::CreateInstance($startupType)
    $startup.cb = [Runtime.InteropServices.Marshal]::SizeOf($startupExType)
    $startup.flags = 0x100
    $startup.input = $inputHandle; $startup.output = $outputHandle; $startup.error = $errorHandle
    $startupEx = [Activator]::CreateInstance($startupExType)
    $startupEx.info = $startup; $startupEx.attributes = $attributes
    $parameters = [object[]]@($application,$command,[IntPtr]::Zero,[IntPtr]::Zero,$true,[uint32]0x80004,[IntPtr]::Zero,$cwd,$startupEx,$process)
    $created = Invoke-Native 'CreateProcessW' $parameters
    $process = $parameters[9]
    Assert-Native $created 'CreateProcessW owned job'
    $resumed = Invoke-Native 'ResumeThread' @($process.thread)
    if ($resumed.Value -eq [uint32]::MaxValue) { throw (New-NativeError $resumed.Error 'ResumeThread') }
    [Console]::Out.WriteLine('AGNOSTIC_READY'); [Console]::Out.Flush()
    $waited = Invoke-Native 'WaitForSingleObject' @($process.process,[uint32]::MaxValue)
    if ($waited.Value -ne 0) { throw (New-NativeError $waited.Error 'WaitForSingleObject') }
    $parameters = [object[]]@($process.process,[uint32]0)
    Assert-Native (Invoke-Native 'GetExitCodeProcess' $parameters) 'GetExitCodeProcess'
    [Console]::Out.WriteLine('AGNOSTIC_EXIT ' + $parameters[1]); [Console]::Out.Flush()
    Assert-Native (Invoke-Native 'TerminateJobObject' @($job,[uint32]1)) 'TerminateJobObject'
  } finally {
    if ($job -ne [IntPtr]::Zero) { [void](Invoke-Native 'CloseHandle' @($job)) }
    if ($process.process -ne [IntPtr]::Zero) {
      [void](Invoke-Native 'TerminateProcess' @($process.process,[uint32]1))
      [void](Invoke-Native 'CloseHandle' @($process.process))
    }
    if ($process.thread -ne [IntPtr]::Zero) { [void](Invoke-Native 'CloseHandle' @($process.thread)) }
    if ($attributesReady) { [void](Invoke-Native 'DeleteProcThreadAttributeList' @($attributes)) }
    foreach ($value in @($attributes,$handles,$jobs)) { if ($value -ne [IntPtr]::Zero) { [Runtime.InteropServices.Marshal]::FreeHGlobal($value) } }
    foreach ($value in @($inputHandle,$outputHandle,$errorHandle)) { if ($value -ne [IntPtr]::Zero) { [void](Invoke-Native 'CloseHandle' @($value)) } }
    if ($nul -ne [IntPtr]::Zero -and $nul -ne [IntPtr](-1)) { [void](Invoke-Native 'CloseHandle' @($nul)) }
  }
} catch {
  [Console]::Error.WriteLine('agnostic-ai Windows launcher: ' + $_.Exception.ToString())
  exit 125
}
