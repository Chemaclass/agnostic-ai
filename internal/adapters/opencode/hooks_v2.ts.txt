import type { Plugin } from "@opencode/plugin"
import { spawn, type ChildProcess } from "node:child_process"
import { existsSync, realpathSync } from "node:fs"
import { delimiter, isAbsolute, join, resolve } from "node:path"
import { fileURLToPath } from "node:url"

const NAME = __NAME__
const COMMANDS: { command: string; args?: string[] }[] = __COMMANDS__
const TIMEOUT = __TIMEOUT__
const ENV: Record<string, string> = __ENV__
const MATCHER = new RegExp(__MATCHER__)
const LABEL = `agnostic-ai hook ${NAME}:`

class GuardInfrastructureError extends Error {}

function windowsExecutable(file: string, cwd: string, env: NodeJS.ProcessEnv) {
  const roots = isAbsolute(file) || /[/\\]/.test(file) ? [resolve(cwd, file)] : [join(cwd, file), ...(env.PATH ?? "").split(delimiter).filter(Boolean).map(root => join(root, file))]
  for (const root of roots) {
    for (const candidate of /\.exe$/i.test(root) ? [root] : [root, root + ".exe"]) {
      if (existsSync(candidate) && /\.exe$/i.test(candidate)) return realpathSync(candidate)
    }
  }
  throw new GuardInfrastructureError(`${LABEL} executable not found: ${file}`)
}

function windowsInput(file: string, args: string[], cwd: string, input: string) {
  const field = (value: string) => {
    const body = Buffer.from(value, "utf8"), size = Buffer.alloc(4)
    if (body.length > 1048576) throw new GuardInfrastructureError(`${LABEL} launcher field exceeds 1 MiB`)
    size.writeInt32LE(body.length); return Buffer.concat([size, body])
  }
  if (args.length > 4096) throw new GuardInfrastructureError(`${LABEL} launcher argument count exceeds 4096`)
  const count = Buffer.alloc(4); count.writeInt32LE(args.length)
  return Buffer.concat([field(file), field(cwd), count, ...args.map(field), Buffer.from(input, "utf8")])
}

function gitBash() {
  const candidates = [
    ...[process.env.ProgramFiles, process.env["ProgramFiles(x86)"]].filter((root): root is string => Boolean(root)).map(root => join(root, "Git", "bin", "bash.exe")),
    ...(process.env.PATH ?? "").split(delimiter).filter(Boolean).map(root => join(root, "bash.exe")),
  ]
  for (const candidate of candidates) {
    if (!existsSync(candidate)) continue
    const actual = realpathSync(candidate)
    if (/[/\\]Git[/\\](?:bin|usr[/\\]bin)[/\\]bash\.exe$/i.test(actual)) return actual
  }
  throw new Error(`${LABEL} shell commands require Git Bash on Windows; no Git Bash executable found`)
}

function launch(command: { command: string; args?: string[] }) {
  if (command.args !== undefined) {
    if (process.platform === "win32" && command.command.endsWith(".sh")) {
      return { file: gitBash(), args: [command.command, ...command.args] }
    }
    return { file: command.command, args: command.args }
  }
  return process.platform === "win32"
    ? { file: gitBash(), args: ["-c", command.command] }
    : { file: "/bin/sh", args: ["-c", command.command] }
}

async function terminate(child: ChildProcess) {
  const pid = child.pid
  if (pid === undefined) return
  if (process.platform !== "win32") {
    try { process.kill(-pid, "SIGKILL") }
    catch (error) {
      if (!(error instanceof Error && "code" in error && error.code === "ESRCH")) throw error
    }
    return
  }
  if (child.exitCode === null && child.signalCode === null && !child.kill("SIGKILL")) throw new GuardInfrastructureError(`${LABEL} could not stop Windows job owner`)
}

const plugin: Plugin.Plugin = {
  id: __ID__,
  async setup(ctx) {
    const active = new Set<(reason: string) => Promise<void>>()
    let unloaded = false
    const run = async (command: { command: string; args?: string[] }, cwd: string, input: string) => {
      if (unloaded) throw new Error(`${LABEL} plugin unloaded`)
      if (cwd.includes("\0")) throw new GuardInfrastructureError(`${LABEL} session working directory must not contain NUL`)
      const started = Date.now()
      const { file, args } = launch(command)
      const windows = process.platform === "win32"
      const env = { ...process.env, ...ENV, AGNOSTIC_AI_TARGET: "opencode" }
      const executable = windows ? windowsExecutable(file, cwd, env) : file
      const payload = windows ? windowsInput(executable, args, cwd, input) : input
      const powershell = join(process.env.SystemRoot ?? "C:\\Windows", "System32", "WindowsPowerShell", "v1.0", "powershell.exe")
      const child = spawn(windows ? powershell : executable, windows ? ["-NoProfile", "-NonInteractive", "-File", fileURLToPath(new URL(__WINDOWS_HELPER__, import.meta.url))] : args, {
        cwd, env,
        detached: !windows, windowsHide: true, stdio: ["pipe", windows ? "pipe" : "ignore", "pipe"],
      })
      let stderr = ""
      let stopped = ""
      let directExit: number | undefined
      let ownerReady = !windows
      let control = ""
      let closed = false
      let killing: Promise<void> | undefined
      const closedPromise = new Promise<void>(resolve => {
        child.once("close", () => { closed = true; resolve() })
      })
      let rejectCompletion: (error: Error) => void = () => {}
      const completion = new Promise<number>((resolve, reject) => {
        rejectCompletion = reject
        child.once("error", reject)
        child.once("close", (code, signal) => {
          if (directExit === 2) resolve(2)
          else if (windows && (!ownerReady || directExit === undefined || code !== 0 || control !== "")) reject(new GuardInfrastructureError(`Windows job owner failed${stopped ? `: ${stopped}` : ""}`))
          else if (signal) reject(new Error(`terminated by ${signal}`))
          else resolve(directExit ?? code ?? 1)
        })
      })
      if (windows) child.stdout?.on("data", (chunk: Buffer) => {
        control += chunk.toString("utf8")
        if (control.length > 1024) { rejectCompletion(new GuardInfrastructureError("Windows job owner control exceeds limit")); return }
        let newline: number
        while ((newline = control.indexOf("\n")) !== -1) {
          const line = control.slice(0, newline).replace(/\r$/, ""); control = control.slice(newline + 1)
          if (line === "AGNOSTIC_READY" && !ownerReady) ownerReady = true
          else if (ownerReady && directExit === undefined && /^AGNOSTIC_EXIT (?:0|[1-9][0-9]*)$/.test(line)) {
            const code = Number(line.slice(14))
            if (Number.isInteger(code) && code >= 0 && code <= 0xffffffff) directExit = code
            else rejectCompletion(new GuardInfrastructureError("invalid Windows job owner exit status"))
          } else rejectCompletion(new GuardInfrastructureError("invalid Windows job owner control"))
        }
      })
      else child.once("exit", code => {
        if (code !== null) directExit = code
        void terminate(child).catch(error => rejectCompletion(error instanceof Error ? error : new Error(String(error))))
      })
      child.stderr?.on("data", (chunk: Buffer) => { stderr = (stderr + chunk.toString("utf8")).slice(-65536) })
      child.stdin?.on("error", error => {
        if (!("code" in error && error.code === "EPIPE")) rejectCompletion(error)
      })
      child.stdin?.end(payload)
      const stop = (reason: string) => {
        stopped ||= reason
        killing ??= (async () => {
          try { await terminate(child) }
          catch (error) { rejectCompletion(error instanceof Error ? error : new Error(String(error))); throw error }
          let timer: ReturnType<typeof setTimeout> | undefined
          try {
            await Promise.race([
              closedPromise,
              new Promise<never>((_resolve, reject) => {
                timer = setTimeout(() => {
                  child.stdin?.destroy(); child.stderr?.destroy(); child.stdout?.destroy()
                  const error = new Error(`${LABEL} process did not close after cleanup`)
                  rejectCompletion(error); reject(error)
                }, 2000)
              }),
            ])
          } finally { clearTimeout(timer) }
        })()
        return killing
      }
      active.add(stop)
      const timer = setTimeout(() => { void stop(`timed out after ${TIMEOUT / 1000} seconds${ownerReady ? "" : " before Windows job owner was ready"}`).catch(() => {}) }, Math.max(0, TIMEOUT - (Date.now() - started)))
      try {
        const code = await completion
        if (stopped && code !== 2) throw new Error(stopped)
        if (code !== 0) console.error(LABEL, command.command, "exited", code, stderr.trim())
        return { code, stderr }
      } catch (error) {
        const detail = stopped || (error instanceof Error ? error.message : String(error))
        if (!closed && child.pid !== undefined) {
          try { await stop(detail) } catch (cleanup) { stderr += `\ncleanup: ${String(cleanup)}` }
        }
        if (directExit === 2) return { code: 2, stderr }
        const message = `${LABEL} ${command.command}: ${detail}${stderr.trim() ? `: ${stderr.trim()}` : ""}`
        if (error instanceof GuardInfrastructureError || (windows && !ownerReady)) throw new GuardInfrastructureError(message)
        throw new Error(message)
      } finally {
        clearTimeout(timer)
        try {
          if (killing) {
            try { await killing } catch (error) {
              if (directExit !== 2) throw error
              console.error(LABEL, "cleanup failed after blocking guard", error)
            }
          }
        } finally { active.delete(stop) }
      }
    }
    const registration = await ctx.tool.hook(__EVENT__, async event => {
      if (unloaded || !MATCHER.test(event.tool)) return
      __AFTER_FILTER__
      let cwd: string
      let input: string
      try {
        const session = await ctx.session.get({ sessionID: event.sessionID })
        cwd = session.location.directory
        input = JSON.stringify(event) + "\n"
      } catch (error) { throw new Error(`${LABEL} resolve session workspace and hook input: ${String(error)}`) }
      for (const command of COMMANDS) {
        let result: { code: number; stderr: string }
        try { result = await run(command, cwd, input) }
        catch (error) {
          console.error(LABEL, error)
          if (unloaded || error instanceof GuardInfrastructureError) throw error
          continue
        }
        __BLOCK__
      }
    })
    return async () => {
      unloaded = true
      const stops = [...active].map(stop => stop("plugin unloaded"))
      const settled = await Promise.allSettled([Promise.resolve().then(() => registration.dispose()), ...stops])
      const failed = settled.find(result => result.status === "rejected")
      if (failed?.status === "rejected") throw failed.reason
    }
  },
}

export default plugin
